const https = require('https');
const http = require('http');

const MIKROTIK_PORT  = parseInt(process.env.MIKROTIK_PORT || '80');
const MIKROTIK_USER  = process.env.MIKROTIK_USER || 'admin';
const MIKROTIK_PASS  = process.env.MIKROTIK_PASS || '';

  console.error('ERROR: RELAY_SECRET is not set in .env. Relay will reject all requests.');

// ── Auth middleware ──────────────────────────────────────────────────────────
function requireSecret(req, res, next) {
  const incoming = req.headers['x-relay-secret'] || '';
  if (!RELAY_SECRET || incoming !== RELAY_SECRET) {
    console.warn(`[${new Date().toISOString()}] Unauthorized request from ${req.ip}`);
    return res.status(401).json({ error: 'Unauthorized' });
  }
  next();
}

// ── MikroTik REST API helper ─────────────────────────────────────────────────
// RouterOS v7 REST API: https://<router>/rest/<path>
// Auth: HTTP Basic (username:password)
function mikrotikRequest(method, path, body = null) {
  return new Promise((resolve, reject) => {
    const auth = Buffer.from(`${MIKROTIK_USER}:${MIKROTIK_PASS}`).toString('base64');
    const options = {
      hostname: MIKROTIK_HOST,
      port: MIKROTIK_PORT,
      path: `/rest${path}`,
      method,
      headers: {
        'Authorization': `Basic ${auth}`,
        'Content-Type': 'application/json',
      },
      // Allow self-signed cert on MikroTik (common for built-in HTTPS)
      rejectUnauthorized: false,
    };

    const transport = MIKROTIK_HTTPS ? https : http;
    const req = transport.request(options, (res) => {
      let data = '';
      res.on('data', chunk => data += chunk);
      res.on('end', () => {
        try {
          resolve({ status: res.statusCode, body: data ? JSON.parse(data) : {} });
        } catch {
          resolve({ status: res.statusCode, body: data });
        }
      });
    });

    req.on('error', reject);
    req.setTimeout(7000, () => { req.destroy(); reject(new Error('MikroTik request timeout')); });

    if (body) req.write(JSON.stringify(body));
    req.end();
  });
}

// ── Route: Find voucher by code ───────────────────────────────────────────────
// Supabase calls this when a code is NOT found in the platform's own DB.
// MikroTik stores vouchers as hotspot users where password = the voucher code.
app.post('/mikrotik/voucher/find', requireSecret, async (req, res) => {
  const { code } = req.body;
  if (!code) return res.status(400).json({ error: 'code is required' });

  try {
    // Query all hotspot users matching this password (code)
    const result = await mikrotikRequest('GET', `/ip/hotspot/user?password=${encodeURIComponent(code)}`);

    console.log(`[find] code="${code}" → status=${result.status} count=${Array.isArray(result.body) ? result.body.length : 0}`);

    if (result.status !== 200 || !Array.isArray(result.body) || result.body.length === 0) {
      return res.json({ user: null });
    }

    // Return the first matching user
    const user = result.body[0];
    return res.json({ user });
  } catch (err) {
    console.error('[find] Error:', err.message);
    return res.status(502).json({ error: 'MikroTik unreachable', detail: err.message });
  }
});

// ── Route: Activate voucher (disable user + bind MAC to active session) ───────
// Called after a valid MikroTik voucher code is confirmed.
app.post('/mikrotik/voucher/activate', requireSecret, async (req, res) => {
  const { userId, macAddress, profile, limitUptime } = req.body;
  if (!userId) return res.status(400).json({ error: 'userId is required' });

  try {
    // Step 1: Disable the voucher user so it can't be used again
    const disableResult = await mikrotikRequest('PATCH', `/ip/hotspot/user/${encodeURIComponent(userId)}`, {
      disabled: 'true',
      comment: `used-${new Date().toISOString().slice(0, 10)}`,
    });
    console.log(`[activate] disable user ${userId} → status=${disableResult.status}`);

    // Step 2: If MAC address provided, bind it to an active session
    // This whitelists the device immediately without needing a login page
    if (macAddress && macAddress !== 'XX:XX:XX:XX:XX:XX') {
      // Remove any existing binding for this MAC first (avoid duplicates)
      const existingBindings = await mikrotikRequest('GET', `/ip/hotspot/active`);
      if (Array.isArray(existingBindings.body)) {
        const existing = existingBindings.body.find(b => b['mac-address'] === macAddress);
        if (existing) {
          await mikrotikRequest('DELETE', `/ip/hotspot/active/${encodeURIComponent(existing['.id'])}`);
          console.log(`[activate] removed existing active session for MAC ${macAddress}`);
        }
      }

      // Add MAC binding (address-list or hotspot active)
      const bindResult = await mikrotikRequest('PUT', `/ip/hotspot/active`, {
        'mac-address': macAddress,
        'server': 'hotspot1',  // adjust if your hotspot server has a different name
        'address': '',
        'uptime': '0s',
        'user': profile || 'default',
      });
      console.log(`[activate] bind MAC ${macAddress} → status=${bindResult.status}`);
    }

    return res.json({ success: true, disabled: disableResult.status === 200 });
  } catch (err) {
    console.error('[activate] Error:', err.message);
    return res.status(502).json({ error: 'MikroTik unreachable', detail: err.message });

# MikroTik Relay — Setup Guide

## What this is
A tiny Node.js server you run on a VPS or Raspberry Pi on your local network.
It acts as a secure bridge between Supabase (cloud) and your MikroTik L009 (LAN).

---

## Why it's needed
Starlink uses CGNAT — your router has no public IP, so Supabase can't reach it directly.
The relay runs on your LAN (or a VPS with a tunnel), opens outbound HTTPS, and forwards
Supabase requests to the router's local REST API.

---

## Quick Start (Raspberry Pi or Ubuntu VPS)

```bash
# 1. Install Node.js (if not already installed)
curl -fsSL https://deb.nodesource.com/setup_20.x | sudo -E bash -
sudo apt-get install -y nodejs

# 2. Copy the relay folder to your device, then:
cd mikrotik-relay
npm install

# 3. Create .env file
cp .env.example .env
nano .env          # fill in RELAY_SECRET and MIKROTIK_PASS

# 4. Start the relay
npm start
```

---

## Making it reachable from the internet (Cloudflare Tunnel — Free)

If running on a Raspberry Pi behind Starlink:

```bash
# Install cloudflared
curl -L https://pkg.cloudflare.com/cloudflare-main.gpg | sudo tee /usr/share/keyrings/cloudflare-main.gpg >/dev/null
echo "deb [signed-by=/usr/share/keyrings/cloudflare-main.gpg] https://pkg.cloudflare.com/cloudflared $(lsb_release -cs) main" | sudo tee /etc/apt/sources.list.d/cloudflared.list
sudo apt-get update && sudo apt-get install cloudflared

# Start tunnel (no account needed for quick tunnel)
cloudflared tunnel --url http://localhost:3000
# Output: https://random-name.trycloudflare.com  ← your Relay URL
```

Paste that URL into **Admin → Settings → MikroTik Relay URL**.

## Local Hosting Setup (Same LAN as MikroTik)

### How It Works Locally

```
Starlink → MikroTik L009 → LAN (192.168.88.0/24)
                │
                ├── Your PC/Laptop/Mini PC  (192.168.88.X)
                │     ├── Portal web server  :5173 or :4173
                │     ├── Relay server       :3000
                │     └── Both run as local services
                │
                └── Customer phones (WiFi hotspot users)

**Key difference from online hosting:** The portal URL will be a local IP instead of a domain name (e.g. `http://192.168.88.10:5173`). Supabase database and Edge Functions still run in the cloud — only the frontend and relay run locally.

| **Paystack webhooks** | Paystack can't reach your local server for payment callbacks | Supabase Edge Functions handle webhooks — these stay in cloud ✓ |
| **PC must stay on** | Portal goes down if PC is off | Use a mini PC (like Intel NUC or Raspberry Pi 4) that stays on 24/7 |
| **Power cuts** | Portal + relay go offline | Add a UPS (battery backup) for the mini PC + router |
| **Starlink outage** | No internet → Paystack payments fail | Voucher redemption still works locally ✓ |
| **LAN IP may change** | Portal URL changes if DHCP reassigns IP | Set a **static LAN IP** for your PC in MikroTik DHCP |

## Set a Static IP for Your PC (Important)

In **Winbox → IP → DHCP Server → Leases**:
- Find your PC in the list
- Right-click → Make Static
- This ensures `192.168.88.10` never changes even after reboots
